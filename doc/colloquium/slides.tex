\begin{frame}{Why binders?}
  \begin{itemize}
    \item Standard canon:
      \begin{itemize}
        \item Church, \emph{The Calculi of Lambda-Conversion} (1941)
        \item de Bruijn, \emph{Lambda calculus notation with nameless dummies} (1972)
        \item Barendregt, \emph{The Lambda Calculus} (1984)
        \item the POPLmark challenge (2005)
        \item \dots
      \end{itemize}
    \item Proof assistants have to pick a representation for terms:
      \begin{itemize}
        \item de Bruijn indices: Rocq, Agda
        \item locally nameless: Isabelle, Lean
        \item named: HOL Light
        \item nominal techniques: Nominal Isabelle
      \end{itemize}
    \item Compilers: global names and linking.
  \end{itemize}
\end{frame}

\begin{frame}{The two papers}
  \textbf{Crole (2012)}, \emph{Alpha equivalence equalities}
  \[ {\simp} = {\simpf} = {\simpi} = {\simr} = {\simrf} \]
  Five definitions of $\alpha$ from the literature are the same relation.

  \vspace{2em}
  \textbf{Norrish and Vestergaard (2007)}, \emph{de Bruijn terms really do work}
  \[ \langle \Lambda_\alpha, \to_\beta \rangle \;\cong\; \langle \mathrm{dB}, \to_{\mathrm{d}} \rangle \]
  Named terms up to $\alpha$ and de Bruijn terms are isomorphic reduction systems.

  \vspace{2em}
  {\small $\Lambda_\alpha$ may quotient by any of the 5; CSLib uses $\simp$.\par}
\end{frame}

\begin{frame}{Permutation-based $\alpha$ (Crole 2012, Defs.~3.1--3.3)}
  \[
    \infrule{atom}{}{a \sim a}
    \qquad
    \infrule{pcg}{M_1 \sim M_1' \quad M_2 \sim M_2'}{M_1\, M_2 \sim M_1'\, M_2'}
  \]
  \[
    \infrule{pi}{\sw{z}{a} \act M \sim \sw{z}{b} \act M'}{\lambda a.\, M \sim \lambda b.\, M'}
  \]
  \bigskip
  \centering
  \small
  \begin{tabular}{@{}llll@{}}
    \toprule
    Relation & rule & side condition & Lean \\
    \midrule
    $\simp$ & $\rn{pi}$ & $z \notin \{a, b\} \cup \vars(M) \cup \vars(M')$
      & \texttt{AlphaEquiv}, \texttt{=$\alpha$} \\
    $\simpf$ & $\rn{pi}_\#$ & $z \notin \{a, b\} \cup \fv(M) \cup \fv(M')$
      & \texttt{AlphaEquivPFresh} \\
    $\simpi$ & $\rn{pi}_1$ & $z \notin \vars(M) \cup \vars(M')$ & \texttt{AlphaEquivP1} \\
    \bottomrule
  \end{tabular}

  \bigskip
  {\footnotesize $\vars$: all atoms, binders included. \ $\fv$: free atoms.
  \ $\sw{z}{a} \act M$: swap $z$ and $a$ everywhere.\par}
\end{frame}

\begin{frame}{Renaming-based $\alpha$ (Crole 2012, Defs.~3.4, 3.5)}
  \[
    \infrule{ref}{}{M \sim M}
    \qquad
    \infrule{sym}{M_1 \sim M_2}{M_2 \sim M_1}
    \qquad
    \infrule{trs}{M_1 \sim M_2 \quad M_2 \sim M_3}{M_1 \sim M_3}
  \]
  \[
    \infrule{pcg}{M_1 \sim M_1' \quad M_2 \sim M_2'}{M_1\, M_2 \sim M_1'\, M_2'}
    \qquad
    \infrule{bcg}{M \sim M'}{\lambda a.\, M \sim \lambda a.\, M'}
  \]
  \[
    \dfrac{}{\lambda a.\, M \sim \lambda a'.\, M[a := a']}\;\alpha
  \]
  \bigskip
  \centering
  \small
  \begin{tabular}{@{}llll@{}}
    \toprule
    Relation & rule & side condition & Lean \\
    \midrule
    $\simr$ & $\alpha$ & $a' \notin \{a\} \cup \vars(M)$ & \texttt{AlphaEquivR} \\
    $\simrf$ & $\alpha_\#$ & $a' \notin \{a\} \cup \fv(M)$ & \texttt{AlphaEquivRFresh} \\
    \bottomrule
  \end{tabular}

  \bigskip
  {\footnotesize $M[a := a']$: capture-avoiding substitution.\par}
\end{frame}

\begin{frame}{$\beta$-reduction on named terms up to $\alpha$}
  $\Lambda_\alpha = \Lambda/{=_\alpha}$, with the operations lifted from representatives:
  \[
    [M]_\alpha\, [N]_\alpha = [M\, N]_\alpha
    \qquad
    \lambda a.\, [M]_\alpha = [\lambda a.\, M]_\alpha
  \]
  \[
    [M]_\alpha[a := [N]_\alpha] = [M[a := N]]_\alpha
  \]
  \medskip
  \[
    \dfrac{}{(\lambda a.\, M)\, N \to_\beta M[a := N]}
    \qquad
    \dfrac{M \to_\beta N}{\lambda a.\, M \to_\beta \lambda a.\, N}
  \]
  \[
    \dfrac{M \to_\beta N}{M\, Z \to_\beta N\, Z}
    \qquad
    \dfrac{M \to_\beta N}{Z\, M \to_\beta Z\, N}
  \]

  \bigskip
  Lean: \texttt{BetaAlpha}, notation $\to_{\beta\alpha}$.
\end{frame}

\begin{frame}{$\beta$-reduction on de Bruijn terms (Nipkow)}
  De Bruijn terms \ $t ::= i \mid t\, t \mid \lambda.\, t$
  \[
    \dfrac{}{(\lambda.\, t)\, u \to_{\mathrm{d}} \mathsf{nsub}\, u\, 0\, t}
    \quad
    \dfrac{t \to_{\mathrm{d}} u}{\lambda.\, t \to_{\mathrm{d}} \lambda.\, u}
    \quad
    \dfrac{t \to_{\mathrm{d}} u}{t\, s \to_{\mathrm{d}} u\, s}
    \quad
    \dfrac{t \to_{\mathrm{d}} u}{s\, t \to_{\mathrm{d}} s\, u}
  \]
  {\footnotesize $\mathsf{nsub}\, u\, k\, t$: replace index $k$ by $u$, decrement the
  free indices above $k$.\par}

  \bigskip
  Lean: \texttt{Beta}, notation $\to_\beta$.
\end{frame}
