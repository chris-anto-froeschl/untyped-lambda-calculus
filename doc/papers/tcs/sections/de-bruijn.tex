\section{An isomorphism with de Bruijn terms}
\label{sec:de-bruijn}

%% The commit of all Lean links is set in leanrev.tex. TODO: re-pin it once the code is upstream.

De Bruijn indices remove the need for alpha equivalence altogether by
replacing bound variable names with numerals counting binder depth.
Norrish and Vestergaard~\cite{norrish2007a} give a direct,
mechanized proof that the pure de Bruijn calculus is isomorphic to
the alpha-quotiented named calculus, so that a result proved on de
Bruijn terms is a result about lambda terms in the ordinary sense. We
formalize this isomorphism for the named terms and the de Bruijn
terms of CSLib; the formalization is under review for inclusion in
CSLib\footnote{\href{https://github.com/leanprover/cslib/pull/\DBPR}{\texttt{https://github.com/leanprover/cslib/pull/\DBPR}}},
on top of the pull requests for the de Bruijn terms (below) and for
Theorem~4.1 of Section~\ref{sec:alpha-equivalences}. By Theorem~\ref{thm:crole-equivalence} it does not
matter which of the five alpha equivalences is quotiented by; the
formalization uses $\simp$.

Norrish and Vestergaard place their isomorphism in a web of directly
proved relationships between ten formalizations of the untyped lambda
calculus, reproduced in Figure~\ref{fig:equivalence-net}. With this
section, CSLib contains the nodes $\LamDB$, $\LamDB'$ and
$\lambda^{N}_{\alpha}$ and the two dashed edges between them.

We keep the notation of~\cite{norrish2007a} on the de Bruijn side,

\subsection{De Bruijn terms}
\label{subsec:db-terms}

The de Bruijn terms, their substitution and $\beta$-reduction, and the
proof of the Church--Rosser theorem used in
Section~\ref{subsec:transfer} are not ours: they are due to Wang, the
syntax and substitution jointly with
Radziwill\footnote{\url{https://github.com/leanprover/cslib/pull/475}}.
%% TODO: PR 475 is open at the time of writing (checked 2026-10-13). The names are the
%% author lines of its files (DeBruijnSyntax.lean: Zayn Wang, Maksym Radziwill; the other three
%% files: Zayn Wang); confirm them and how to cite.

\begin{definition}[De Bruijn terms {\cite[Section~2.2]{norrish2007a}}]
\label{def:db-terms}
\lean{\DBSrc/Unscoped/Untyped/DeBruijnSyntax.lean\#L56}
The set $\LamDB$ of de Bruijn terms is generated by
\[
  t \;::=\; \dV\, i \mid \dAPP\, t\, t \mid \dABS\, t
  \qquad (i \in \mathbb{N}).
\]
Lifting
\lean{\DBSrc/NamedDeBruijn/PureDeBruijn.lean\#L69}
and substitution
\lean{\DBSrc/NamedDeBruijn/PureDeBruijn.lean\#L85}
are defined as in Nipkow's formalization~\cite{nipkow2001a}:
\begin{align*}
  \lift\, (\dV\, i)\, k &=
    \text{if } i < k \text{ then } \dV\, i \text{ else } \dV\, (i + 1)\\
  \lift\, (\dAPP\, t\, u)\, k &= \dAPP\, (\lift\, t\, k)\, (\lift\, u\, k)\\
  \lift\, (\dABS\, t)\, k &= \dABS\, (\lift\, t\, (k + 1))\\[1ex]
  \nsub\, s\, k\, (\dV\, i) &=
    \text{if } k < i \text{ then } \dV\, (i - 1)
    \text{ else if } i = k \text{ then } s \text{ else } \dV\, i\\
  \nsub\, s\, k\, (\dAPP\, t\, u) &= \dAPP\, (\nsub\, s\, k\, t)\, (\nsub\, s\, k\, u)\\
  \nsub\, s\, k\, (\dABS\, t) &= \dABS\, (\nsub\, (\lift\, s\, 0)\, (k + 1)\, t)
\end{align*}
\end{definition}

The function $\nsub$ substitutes for the index $k$ and at the same
time decrements all larger free indices, which is what
$\beta$-contraction needs when a binder disappears.

\begin{definition}[{\cite[Definition~1]{norrish2007a}}]
\label{def:db-beta}
\lean{\DBSrc/Unscoped/Untyped/BetaReduction.lean\#L41}
The relation $\tod$ on $\LamDB$ is the least relation satisfying
\begin{gather*}
  \infrule{}{}{\dAPP\, (\dABS\, t)\, u \tod \nsub\, u\, 0\, t}
  \qquad
  \infrule{}{t \tod u}{\dABS\, t \tod \dABS\, u}
  \\[1ex]
  \infrule{}{t \tod u}{\dAPP\, t\, s \tod \dAPP\, u\, s}
  \qquad
  \infrule{}{t \tod u}{\dAPP\, s\, t \tod \dAPP\, s\, u}.
\end{gather*}
\end{definition}

\subsection{The alpha-quotiented calculus}
\label{subsec:lam-alpha}

\begin{definition}[{\cite[Section~2.1]{norrish2007a}}]
\label{def:lam-alpha}
\lean{\DBSrc/Named/Untyped/AlphaQuotient.lean\#L71}
$\LamA$ is the quotient of $\Lam$ by $\simp$. We write $\cls{M}$ for
the class of a term $M \in \Lam$. Application, abstraction and
substitution
\lean{\DBSrc/Named/Untyped/AlphaQuotient.lean\#L99}
are lifted from $\Lam$ to $\LamA$:
\[
  \cls{M}\, \cls{N} = \cls{M\, N}
  \qquad
  \lam{a}{\cls{M}} = \cls{\lam{a}{M}}
  \qquad
  \subst{\cls{M}}{a}{\cls{N}} = \cls{\subst{M}{a}{N}}.
\]
\end{definition}

These operations are well defined because $\simp$ is a congruence and
is preserved by substitution
\lean{\CroleSrc/Named/Untyped/AlphaEquivProperties.lean\#L378}.
Abstraction on $\LamA$ is not injective
(Example~\ref{ex:lam-alpha}), so $\LamA$ is not a free algebra.

\begin{example}
\label{ex:lam-alpha}
\lean{\DemoSrc\#L110}
For distinct atoms $a, b, c$,
$\lam{a}{\cls{a\,b}} = \lam{c}{\cls{c\,b}}$
(Example~\ref{ex:five-relations}), although
$\cls{a\,b} \neq \cls{c\,b}$.
\end{example}

\begin{definition}[$\beta$-reduction on $\LamA$]
\label{def:beta-alpha}
\lean{\DBSrc/Named/Untyped/AlphaQuotient.lean\#L122}
The relation $\tobeta$ on $\LamA$ is the least relation satisfying
\begin{gather*}
  \infrule{}{}{(\lam{a}{M})\, N \tobeta \subst{M}{a}{N}}
  \qquad
  \infrule{}{M \tobeta N}{\lam{a}{M} \tobeta \lam{a}{N}}
  \\[1ex]
  \infrule{}{M \tobeta N}{M\, Z \tobeta N\, Z}
  \qquad
  \infrule{}{M \tobeta N}{Z\, M \tobeta Z\, N}.
\end{gather*}
\end{definition}

This is the system $\lambda^{\mathrm{N}}_{\alpha}$
of~\cite{norrish2007a}: substitution and reduction are defined on the
quotient itself and not on raw terms.

Two systems $\langle X, \to_X \rangle$ and $\langle Y, \to_Y \rangle$
are \emph{isomorphic} if there is a bijection $f \colon X \to Y$ with
$x_1 \to_X x_2 \iff f(x_1) \to_Y f(x_2)$ for all
$x_1, x_2$~\cite[Section~2.3]{norrish2007a}.

\subsection{The isomorphism theorem}
\label{subsec:isomorphism}

\begin{theorem}[{\cite[Theorem~20]{norrish2007a}}]
\label{thm:db-isomorphism}
\lean{\DBSrc/NamedDeBruijn/Isomorphism.lean\#L146}
The systems $\langle \LamA, \tobeta \rangle$ and
$\langle \LamDB, \tod \rangle$ are isomorphic, as witnessed by the
bijection $f$ of Theorem~\ref{thm:db-bijection} below: for all
$M, N \in \LamA$,
\[
  M \tobeta N \iff f(M) \tod f(N).
\]
\end{theorem}

\begin{proof}
From Theorems~\ref{thm:beta-dbeta-prime}
and~\ref{thm:dbeta-coincide} below.
\end{proof}

The notion of isomorphic reduction systems used here is exactly
Mathlib's relation isomorphism, and the formalization also states
Theorem~\ref{thm:db-isomorphism} in that form,
$\langle \LamA, \tobeta \rangle \simeq_r \langle \LamDB, \tod \rangle$
\lean{\DBSrc/NamedDeBruijn/Isomorphism.lean\#L151}.

The proof of Norrish and Vestergaard goes through an intermediate
formalism (Figure~\ref{fig:route}).
Section~\ref{subsec:db-prime} gives a second set of rules on the de
Bruijn terms that is built to resemble $\LamA$.
Section~\ref{subsec:bijection} constructs $f$ and shows that it turns
$\tobeta$ into these rules, and Section~\ref{subsec:dbeta} shows that
they define the same relation as $\tod$.

\subsection{An intermediate formalism}
\label{subsec:db-prime}

The proof goes through a second presentation $\LamDB'$ of the same
type $\LamDB$ that is built to resemble $\LamA$. It replaces $\nsub$
by a substitution that leaves the other indices alone, and $\dABS$ by
an abstraction over a chosen index.

\begin{definition}[{\cite[Definitions~4 and~5]{norrish2007a}}]
\label{def:dlam}
\lean{\DBSrc/NamedDeBruijn/PureDeBruijn.lean\#L106}
\lean{\DBSrc/NamedDeBruijn/PureDeBruijn.lean\#L215}
\begin{align*}
  \dbsub\, s\, k\, (\dV\, i) &= \text{if } i = k \text{ then } s \text{ else } \dV\, i\\
  \dbsub\, s\, k\, (\dAPP\, t\, u) &= \dAPP\, (\dbsub\, s\, k\, t)\, (\dbsub\, s\, k\, u)\\
  \dbsub\, s\, k\, (\dABS\, t) &= \dABS\, (\dbsub\, (\lift\, s\, 0)\, (k + 1)\, t)\\[1ex]
  \dLAM\, i\, t &= \dABS\, (\dbsub\, (\dV\, 0)\, (i + 1)\, (\lift\, t\, 0))
\end{align*}
\end{definition}

Like abstraction on $\LamA$, $\dLAM$ is not injective
(Example~\ref{ex:dlam}). Every $\dABS$
term is a $\dLAM$ term over any index that is not free in it
\lean{\DBSrc/NamedDeBruijn/PureDeBruijn.lean\#L264}
\cite[Lemma~6]{norrish2007a}.

\begin{example}
\label{ex:dlam}
\lean{\DemoSrc\#L272}
For every index $i$,
\[
  \dLAM\, i\, (\dV\, i)
  = \dABS\, (\dbsub\, (\dV\, 0)\, (i + 1)\, (\dV\, (i + 1)))
  = \dABS\, (\dV\, 0),
\]
the de Bruijn form of $\lam{a}{a}$ for every atom $a$.
\end{example}

\begin{definition}[{\cite[Definition~7]{norrish2007a}}]
\label{def:dbeta-prime}
\lean{\DBSrc/NamedDeBruijn/PureDeBruijn.lean\#L319}
The relation $\todp$ on $\LamDB$ is the least relation satisfying
\begin{gather*}
  \infrule{}{}{\dAPP\, (\dLAM\, i\, t)\, u \todp \dbsub\, u\, i\, t}
  \qquad
  \infrule{}{t \todp u}{\dLAM\, i\, t \todp \dLAM\, i\, u}
  \\[1ex]
  \infrule{}{t \todp u}{\dAPP\, t\, s \todp \dAPP\, u\, s}
  \qquad
  \infrule{}{t \todp u}{\dAPP\, s\, t \todp \dAPP\, s\, u}.
\end{gather*}
\end{definition}

\subsection{The bijection}
\label{subsec:bijection}

Fix a bijection between $\Atoms$ and $\mathbb{N}$, written with a hat
in both directions as in~\cite{norrish2007a}: $\hat{a}$ is the number
of the atom $a$ and $\hat{\imath}$ the atom with number $i$.

\begin{theorem}[{\cite[Theorem~11]{norrish2007a}}]
\label{thm:db-bijection}
\lean{\DBSrc/NamedDeBruijn/FromTerm.lean\#L363}
There is a bijection $f \colon \LamA \to \LamDB$ with
\begin{equation}
\label{eq:f}
  f(\cls{a}) = \dV\, \hat{a}
  \qquad
  f(M\, N) = \dAPP\, (f\, M)\, (f\, N)
  \qquad
  f(\lam{a}{M}) = \dLAM\, \hat{a}\, (f\, M).
\end{equation}
\end{theorem}

\begin{example}[Running example, continued]
\label{ex:f}
\lean{\DemoSrc\#L224}
For distinct atoms $a$ and $b$,
\[
  f(\lam{a}{a\,b})
  = \dLAM\, \hat{a}\, (\dAPP\, (\dV\, \hat{a})\, (\dV\, \hat{b}))
  = \dABS\, (\dAPP\, (\dV\, 0)\, (\dV\, (\hat{b} + 1))):
\]
the bound atom becomes the index $0$, and the free atom $b$ is
shifted past the binder. For every atom $c \neq b$, the
alpha-equivalent term $\lam{c}{c\,b}$ has the same image
\lean{\DemoSrc\#L236}.
\end{example}

Since $\LamA$ is not a free algebra, the equations~\eqref{eq:f} do not
define a function by themselves. Norrish and Vestergaard obtain $f$
from Pitts' recursion principle for
$\LamA$~\cite{pitts2006a}, for which $\LamDB$ has to be shown to be
a nominal set. The formalization takes a more elementary route: the
equations~\eqref{eq:f} define a function on raw terms by structural
recursion
\lean{\DBSrc/NamedDeBruijn/FromTerm.lean\#L106},
this function is shown to respect $\simp$
\lean{\DBSrc/NamedDeBruijn/FromTerm.lean\#L181},
and $f$ is its lift to the quotient
\lean{\DBSrc/NamedDeBruijn/FromTerm.lean\#L339}.
Injectivity
\lean{\DBSrc/NamedDeBruijn/FromTerm.lean\#L192}
is proved by structural induction on raw terms. Its abstraction case
rests on the fact that $\dLAM$ is injective up to a transposition:
$\dLAM\, i\, t = \dLAM\, j\, u$ holds exactly if $i = j$ and $t = u$,
or $i \neq j$, the index $i$ is not free in $u$, and $t$ is $u$ with
$i$ and $j$ exchanged
\lean{\DBSrc/NamedDeBruijn/LiftingPermutation.lean\#L332}.
Surjectivity
\lean{\DBSrc/NamedDeBruijn/FromTerm.lean\#L263}
is proved, as in~\cite{norrish2007a}, by complete induction on the size
of the de Bruijn term, using Lemma~6 of~\cite{norrish2007a} for
$\dABS$ terms.

\begin{theorem}[{\cite[Theorem~12]{norrish2007a}}]
\label{thm:beta-dbeta-prime}
\lean{\DBSrc/NamedDeBruijn/FromTerm.lean\#L444}
For all $M, N \in \LamA$: $M \tobeta N$ if and only if
$f(M) \todp f(N)$.
\end{theorem}

\begin{proof}[Proof idea]
The bijection turns substitution on $\LamA$ into the index-preserving
substitution $\dbsub$,
$f(\subst{M}{a}{N}) = \dbsub\, (f\, N)\, \hat{a}\, (f\, M)$
\lean{\DBSrc/NamedDeBruijn/FromTerm.lean\#L231}, and abstraction
into $\dLAM$, so that each rule of $\tobeta$ becomes the
corresponding rule of $\todp$~\cite[proof of
Theorem~12]{norrish2007a}.
\end{proof}


\subsection{From $\todp$ to $\tod$}
\label{subsec:dbeta}

It remains to show that $\tod$ and $\todp$ are the same relation on
$\LamDB$. Their rules for redexes are related by an identity between
$\dbsub$ and $\nsub$
\lean{\DBSrc/NamedDeBruijn/LiftingPermutation.lean\#L54}
\cite[Lemma~13]{norrish2007a}; the difficulty is
to match the congruence rule for $\dABS$ with the one for $\dLAM$.
This is where permutations enter on the de Bruijn side. A permutation
$\pi$ of atoms acts on a de Bruijn term through the free indices
\lean{\DBSrc/NamedDeBruijn/DeBruijnPerm.lean\#L169}:
\[
  \pi \act (\dV\, i) = \dV\, \widehat{\pi(\hat{\imath})}
  \qquad
  \pi \act (\dAPP\, t\, u) = \dAPP\, (\pi \act t)\, (\pi \act u)
  \qquad
  \pi \act (\dABS\, t) = \dABS\, (\inc(\pi) \act t),
\]
where $\inc(\pi)$ is $\pi$ shifted by one index, so that it fixes the
index bound by the abstraction~\cite[Definition~8]{norrish2007a}.

\begin{example}
\label{ex:dpm}
\lean{\DemoSrc\#L277}
Let $\hat{a} = 0$ and $\hat{b} = 1$. The term $\dABS\, (\dV\, 1)$ is
$\lam{x}{a}$, since index $1$ under one binder is the free index $0$.
Under the binder, $\inc\sw{a}{b}$ exchanges the indices $1$ and $2$,
so
\[
  \sw{a}{b} \act \dABS\, (\dV\, 1) = \dABS\, (\dV\, 2),
\]
which is $\lam{x}{b}$. Exchanging $0$ and $1$ under the binder
instead would give $\dABS\, (\dV\, 0)$, which is $\lam{x}{x}$
\lean{\DemoSrc\#L280}.
\end{example}

A relation $R$ on $\LamDB$ is \emph{equivariant} if
$R\,(\pi \act t)\,(\pi \act u) \iff R\, t\, u$ for all $\pi$.

By~\cite[Theorem~10]{norrish2007a}, $\dbsub$ and $\dLAM$ commute
with permutations, which act on their index arguments $j$ as
$\widehat{\pi(\hat{\jmath})}$
\lean{\DBSrc/NamedDeBruijn/DeBruijnPerm.lean\#L291}%
\lean{\DBSrc/NamedDeBruijn/DeBruijnPerm.lean\#L308}, and hence
$\todp$ is equivariant
\lean{\DBSrc/NamedDeBruijn/DeBruijnPerm.lean\#L328}.

The key observation of~\cite{norrish2007a} is that lifting is itself
the action of a permutation
\lean{\DBSrc/NamedDeBruijn/LiftingPermutation.lean\#L144}
\cite[Lemma~14]{norrish2007a}, and so is the substitution of a fresh
index
\lean{\DBSrc/NamedDeBruijn/LiftingPermutation.lean\#L200}.
A $\dABS$ term and its presentation as a $\dLAM$ term therefore
differ by a permutation, which an equivariant relation cannot see.

\begin{theorem}[{\cite[Theorem~15]{norrish2007a}}]
\label{thm:abs-congr}
\lean{\DBSrc/NamedDeBruijn/LiftingPermutation.lean\#L358}
Let $R$ be an equivariant relation on $\LamDB$. Then
\[
  \bigl(\forall t\, u.\; R\, t\, u \Rightarrow
     R\,(\dABS\, t)\,(\dABS\, u)\bigr)
  \iff
  \bigl(\forall i\, t\, u.\; R\, t\, u \Rightarrow
     R\,(\dLAM\, i\, t)\,(\dLAM\, i\, u)\bigr).
\]
\end{theorem}

Theorem~\ref{thm:abs-congr} applies to $\todp$, which is equivariant. To apply it to $\tod$ one needs
equivariance of $\tod$, and hence the behavior of $\nsub$ under a
permutation
\lean{\DBSrc/NamedDeBruijn/Isomorphism.lean\#L61}
\cite[Lemma~17]{norrish2007a}.

\begin{theorem}[{\cite[Theorem~19]{norrish2007a}}]
\label{thm:dbeta-coincide}
\lean{\DBSrc/NamedDeBruijn/Isomorphism.lean\#L126}
For all $t, u \in \LamDB$: $t \todp u$ if and only if $t \tod u$.
\end{theorem}

\begin{example}[Running example, continued]
\label{ex:beta}
For distinct atoms $a, b, c$, the step
$(\lam{a}{\cls{a\,b}})\,\cls{c} \tobeta \cls{c\,b}$
\lean{\DemoSrc\#L254} becomes
\[
  \dAPP\, (\dABS\, (\dAPP\, (\dV\, 0)\, (\dV\, (\hat{b} + 1))))\,
    (\dV\, \hat{c})
  \;\tod\;
  \dAPP\, (\dV\, \hat{c})\, (\dV\, \hat{b})
\]
\lean{\DemoSrc\#L264}: $\nsub$ replaces the bound index $0$ by
$\dV\, \hat{c}$ and decrements the index of $b$, whose binder has
disappeared.
\end{example}

\subsection{Using the isomorphism}
\label{subsec:transfer}

An isomorphism of reduction systems transports every property that is
stated in terms of the reduction relation. The de Bruijn development
we build on proves the Church--Rosser theorem for $\tod$
\lean{\DBSrc/Unscoped/Untyped/ChurchRosser.lean\#L45}.
With Theorem~\ref{thm:db-isomorphism} it becomes a theorem about
named terms up to alpha equivalence, with no further work on the
named side.

\begin{corollary}[Church--Rosser for $\LamA$]
\label{cor:church-rosser}
\lean{\DBSrc/NamedDeBruijn/Transfer.lean\#L147}
The relation $\tobeta$ on $\LamA$ is confluent: if $M \tobetastar N_1$
and $M \tobetastar N_2$ then there is $P$ with $N_1 \tobetastar P$ and
$N_2 \tobetastar P$.
\end{corollary}

\begin{corollary}
\label{cor:normal-form-unique}
\lean{\DBSrc/NamedDeBruijn/Transfer.lean\#L220}
If $M \tobetastar N_1$ and $M \tobetastar N_2$ and both $N_1$ and
$N_2$ are $\beta$-normal, then $N_1 = N_2$.
\end{corollary}

\subsection{Notes on the formalization}
\label{subsec:db-formalization-notes}

\begin{itemize}
  \item \emph{Atoms.} Sections~\ref{sec:cslib}
    and~\ref{sec:alpha-equivalences} are parametric in $\Atoms$. The
    isomorphism is stated for one concrete type of atoms, the natural
    numbers wrapped in a structure, for which the two hat maps are the
    projection and the constructor
    \lean{\LeanRepo/blob/\LeanRev/Cslib/Logics/Nominal/Atom.lean\#L44}.
    %% TODO: the types of atoms and of finitely supported permutations (Atom, FinPerm)
    %% are ported from the repository spitters/nominal-lean (Bas Spitters), which in turn ports
    %% the nominal layer of SSProve. Decide how to credit and cite this.
  \item \emph{Permutations.} Norrish and Vestergaard represent a
    permutation as a list of pairs of strings. The formalization uses
    finitely supported bijections on atoms
    \lean{\LeanRepo/blob/\LeanRev/Cslib/Logics/Nominal/FinPerm.lean\#L56}.
    Accordingly $\inc$ is not defined by recursion over a list but as
    the permutation that fixes the atom with number $0$ and otherwise
    acts as $\pi$ with all numbers shifted by one
    \lean{\DBSrc/NamedDeBruijn/DeBruijnPerm.lean\#L113}.
  \item \emph{Substitution on $\LamDB$.} CSLib does not define $\nsub$
    by the equations of Definition~\ref{def:db-terms}, but as a
    composition of a lifting, the substitution $\dbsub$ and a lowering
    of indices. The equations of Definition~\ref{def:db-terms} are
    proved as lemmas.
  \item \emph{The bijection.} $f$ is a bijection in the sense of an
    explicit pair of mutually inverse functions. Its inverse is
    obtained from surjectivity by choice and is not computable.
  \item \emph{Equivariance.} There is no general notion of an
    equivariant relation in the library used. In
    Theorem~\ref{thm:abs-congr} equivariance is a hypothesis on $R$,
    and it is proved separately for each of the relations to which the
    theorem is applied.
\end{itemize}
