\section{Characterizations of \texorpdfstring{$\alpha$}{α}-equivalence}
\label{sec:alpha-equivalences}

This section presents Crole's result~\cite{crole2012a} as it is
formalized in CSLib. We keep Crole's names for the relations, the rules
and the side conditions, and each statement references its counterpart
in~\cite{crole2012a}.

\subsection{The relations}
\label{subsec:five-definitions}

The first three relations are syntax-directed and differ only in the
side condition on the atom $z$ that replaces the two bound atoms.

\begin{definition}[{\cite[Definitions~3.1--3.3]{crole2012a}}]
\label{def:simp}
The relation $\simp$
\lean{\CroleSrc/Named/Untyped/Basic.lean\#L82}
on $\Lam$ is inductively defined by
\begin{gather*}
  \infrule{atom}{}{a \simp a}
  \qquad
  \infrule{pcg}{M_1 \simp M_1' \quad M_2 \simp M_2'}
               {M_1\, M_2 \simp M_1'\, M_2'}
  \qquad
  \infrule{pi}{\sw{z}{a} \act M \simp \sw{z}{b} \act M'}
              {\lam{a}{M} \simp \lam{b}{M'}}
  \; [z \nocc a, b, M, M'].
\end{gather*}
The relations $\simpf$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Defs.lean\#L92}
and $\simpi{1}$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Defs.lean\#L110}
are defined by the same rules, with the side condition of \textsc{pi}
replaced by
\[
  [z \fresh a, b, M, M']
  \qquad\text{and}\qquad
  [z \nocc M, M'],
\]
respectively. The resulting rules are called \textsc{pi\#} and \textsc{pi1} respectively.
\end{definition}

Read as rules on $\lambda$-terms, \textsc{atom} and \textsc{pcg} compare
variables and applications component by component, and \textsc{pi}
compares two abstractions by moving both bound atoms to a common unused
atom $z$ and comparing the bodies.

The relation $\simp$ follows Gabbay and Pitts~\cite{gabbay2002a} and
is the $\alpha$-equivalence $\alphaeq$ that CSLib already provides. The remaining two relations
are generated by the renaming axiom and are congruences by
definition.

\begin{definition}[{\cite[Definitions~3.4 and~3.5]{crole2012a}}]
\label{def:simr}
The relation $\simr$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Defs.lean\#L153}
on $\Lam$ is inductively defined by
\begin{gather*}
  \infrule{ref}{}{M \simr M}
  \qquad
  \infrule{sym}{M_1 \simr M_2}{M_2 \simr M_1}
  \qquad
  \infrule{trs}{M_1 \simr M_2 \quad M_2 \simr M_3}{M_1 \simr M_3}
  \\[1ex]
  \infrule{pcg}{M_1 \simr M_1' \quad M_2 \simr M_2'}
               {M_1\, M_2 \simr M_1'\, M_2'}
  \qquad
  \infrule{bcg}{M \simr M'}{\lam{a}{M} \simr \lam{a}{M'}}
  \qquad
  \infrule{$\alpha$}{}{\lam{a}{M} \simr \lam{a'}{\subst{M}{a}{a'}}}
  \; [a' \nocc a, M].
\end{gather*}
The relation $\simrf$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Defs.lean\#L173}
is defined by the same rules, with the side condition of $\alpha$
replaced by $[a' \fresh a, M]$. The resulting rule is called $\alpha\#$.
\end{definition}

The relation $\simr$ corresponds to the congruence $\equiv_\alpha$
of Barendregt~\cite[Definition~2.1.11]{barendregt1985a}, and $\simrf$
to that of Hindley and Seldin~\cite[Definition~1.17]{hindley2008a}.

\begin{example}[Running example]
\label{ex:five-relations}
For distinct atoms $a, b, c$, the terms $\lam{a}{a\,b}$ and
$\lam{c}{c\,b}$ are related by all of these relations, each time by a
single application of the abstraction rule.
\begin{itemize}
  \item \textsc{pi}
    \lean{\DemoSrc\#L110},
    \textsc{pi\#}
    \lean{\DemoSrc\#L127}
    and \textsc{pi1}
    \lean{\DemoSrc\#L135}: any witness $d$ distinct from $a, b, c$
    meets all three side conditions, and
    $\sw{d}{a} \act (a\,b) = d\,b = \sw{d}{c} \act (c\,b)$.
  \item $\alpha$
    \lean{\DemoSrc\#L149}
    and $\alpha\#$
    \lean{\DemoSrc\#L155}: rename $a$ to $c$, which does not occur in
    $a\,b$. Then $\subst{(a\,b)}{a}{c} = c\,b$.
\end{itemize}
\end{example}

\subsection{The characterization theorem}
\label{subsec:equivalence-theorem}

\begin{theorem}[Crole {\cite[Theorems~4.1, 4.2, 4.4--4.6]{crole2012a}}]
\label{thm:crole-equivalence}
For all terms $M, N \in \Lam$ the following are equivalent:
\begin{enumerate}
  \item[(i)] $M \simp N$,
  \item[(ii)] $M \simpf N$
    \lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Equalities.lean\#L99},
  \item[(iii)] $M \simpi{1} N$
    \lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Equalities.lean\#L140},
  \item[(iv)] $M \simr N$
    \lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Equalities.lean\#L212},
  \item[(v)] $M \simrf N$
    \lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Equalities.lean\#L251}.
\end{enumerate}
\end{theorem}

The proof is outlined in Section~\ref{subsec:key-lemmas}. The side
condition of \textsc{pi} cannot be weakened much further.

\begin{proposition}[{\cite[Proposition~4.3]{crole2012a}}]
\label{prop:variants}
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Equalities.lean\#L146}
Let $\simpi{2}$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Defs.lean\#L127}
and $\simpi{3}$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Defs.lean\#L139}
be defined as $\simp$, with the side condition of \textsc{pi} replaced
by $[z \nocc a, b]$ and by no condition, respectively. Then
${\simpi{2}} \neq {\simp}$ and ${\simpi{3}} \neq {\simp}$.
\end{proposition}

\begin{proof}
For distinct atoms $a, b, z$ we have
$\lam{a}{a\, z} \simpi{2} \lam{b}{b\, a}$
\lean{\DemoSrc\#L161} with witness $z$, and
$\lam{a}{b\, a} \simpi{3} \lam{b}{a\, b}$
\lean{\DemoSrc\#L179} with witness $b$. In both cases the two
terms have different free atoms. Since $\simp$ preserves free atoms
\lean{\CroleSrc/Named/Untyped/AlphaEquivProperties.lean\#L56},
they are not related by $\simp$
\lean{\DemoSrc\#L170}
\lean{\DemoSrc\#L186}.
\end{proof}

\subsection{The proof}
\label{subsec:key-lemmas}

Since $\simp$ has no rules for reflexivity, symmetry and
transitivity, these have to be proved. CSLib already provides
them~\lean{\CroleSrc/Named/Untyped/AlphaEquivProperties.lean\#L70}%
\lean{\CroleSrc/Named/Untyped/AlphaEquivProperties.lean\#L82}%
\lean{\CroleSrc/Named/Untyped/AlphaEquivProperties.lean\#L140},
together with two facts used below: related terms have the same free
atoms~\lean{\CroleSrc/Named/Untyped/AlphaEquivProperties.lean\#L56}, and
substitution respects $\simp$: if $M \simp M'$ and $N \simp N'$ then
$\subst{M}{a}{N} \simp \subst{M'}{a}{N'}$
\lean{\CroleSrc/Named/Untyped/AlphaEquivProperties.lean\#L378}.
Beyond these, the proof of
Theorem~\ref{thm:crole-equivalence} rests on three facts: $\simp$ is
preserved by swapping, a permutation acts up to $\simp$ only through
the free atoms, and swapping with an unused atom is, up to
$\alpha$-equivalence, the same as substitution.

\begin{lemma}[{\cite[Lemma~6.1]{crole2012a}}]
\label{lem:swap-preserve}
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Swap.lean\#L285}
If $M \simp M'$ then $\sw{u}{v} \act M \simp \sw{u}{v} \act M'$, for
all atoms $u, v$.
\end{lemma}

This is an instance of the equivariance principle of nominal
techniques~\cite[Theorem~4.4]{gabbay2011a}, which will again appear in Section~\ref{sec:de-bruijn}.
For the rule \textsc{pi}, each of the atoms $a$, $b$, $z$ of the rule
is equal to $u$, equal to $v$, or different from both, which gives
$3^3 = 27$ cases. The side condition $z \neq a, b$ excludes 10 of
them. In the 9 cases with $z \neq u, v$ the swap $\sw{u}{v}$ commutes
with $\sw{z}{a}$ and $\sw{z}{b}$, and $z$ is again a witness. Crole
works out one of these and three of the remaining
eight~\cite[pp.~14--15]{crole2012a}. The formalization derives the
other five by exchanging $u$ and $v$ or the two sides of $\simp$
(Table~\ref{tab:lemma61}).

\begin{lemma}[{\cite[Lemma~6.2]{crole2012a}}]
\label{lem:agreement}
Let $\pi, \pi'$ be permutations of $\Atoms$ and $M$ a term.
\begin{enumerate}
  \item \lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Swap.lean\#L91}
    If $\occ(M) \subseteq \AS(\pi, \pi')$ then
    $\pi \act M = \pi' \act M$.
  \item \lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/Swap.lean\#L379}
    If $\free(M) \subseteq \AS(\pi, \pi')$ then
    $\pi \act M \simp \pi' \act M$.
\end{enumerate}
\end{lemma}

A permutation thus acts on a term only through the atoms of the term,
and up to $\alpha$-equivalence only through its free atoms.

\begin{example}
\label{ex:agreement}
Let $M = \lam{a}{a\,b}$, $\pi = \sw{a}{c}$ and $\pi' = \mathrm{id}$
with $c \neq b$. Part~1 of Lemma~\ref{lem:agreement} does not apply,
since $\pi$ and $\pi'$ disagree on $a \in \occ(M)$, and indeed
$\pi \act M = \lam{c}{c\,b}$ is a different term from $M$. But
$\free(M) = \{b\} \subseteq \AS(\pi, \pi')$, so part~2 gives
$\lam{c}{c\,b} \simp \lam{a}{a\,b}$, the running example of
Example~\ref{ex:five-relations}
\lean{\DemoSrc\#L110}.
\end{example}

The third fact links the two families of relations. Crole points
out that the proof sketch of Gabbay and
Pitts~\cite[Proposition~2.2]{gabbay2002a} lacks it: they argue only
that swapping preserves $\simr$, which gives no connection between
swapping and substitution~\cite[p.~5]{crole2012a}.

\begin{lemma}[{\cite[Lemmas~6.3 and~6.4]{crole2012a}}]
\label{lem:swap-subst}
If $z \nocc M$ then
$\sw{z}{a} \act M \simr \subst{M}{a}{z}$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/SwapSubst.lean\#L60}
and
$\sw{z}{a} \act M \simrf \subst{M}{a}{z}$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/SwapSubst.lean\#L85}.
\end{lemma}

\begin{example}
\label{ex:swap-subst}
The two sides of Lemma~\ref{lem:swap-subst} differ on bound
occurrences of $a$, which the swap replaces and the substitution does
not. Let $M = a\,(\lam{a}{a})$ and $z \neq a$. Then
\[
  \sw{z}{a} \act M = z\,(\lam{z}{z})\text{\lean{\DemoSrc\#L196}}
  \qquad\text{and}\qquad
  \subst{M}{a}{z} = z\,(\lam{a}{a})\text{\lean{\DemoSrc\#L200}},
\]
and the two are related by \textsc{pcg}, $\alpha$ and \textsc{sym}
\lean{\DemoSrc\#L205}.
\end{example}

For $\simp$ the corresponding statement compares two swaps. We need
only the case $n = 1$ of Crole's lemma.

\begin{lemma}[{\cite[Lemma~6.5]{crole2012a}, case $n = 1$}]
\label{lem:swap-chain}
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/SwapSubst.lean\#L246}
Let $z, a, a'$ be distinct atoms and $M$ a term with $z \fresh M$ and
$a' \fresh M$. Then
$\sw{z}{a} \act M \simp \sw{z}{a'} \act \subst{M}{a}{a'}$.
\end{lemma}

\begin{proof}[Proof idea]
Crole omits the proof. First
$\subst{M}{a}{a'} \simp \sw{a}{a'} \act M$
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/SwapSubst.lean\#L154}:
rename the bound
occurrences of $a'$ away~\cite[Lemma~6.6]{crole2012a}, use that
substitution respects $\simp$, and apply the $\simp$-form of
Lemma~\ref{lem:swap-subst}
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/SwapSubst.lean\#L135}.
Then $\sw{z}{a}$ and
$\sw{z}{a'} \circ \sw{a}{a'}$ agree on $\free(M)$, and
Lemma~\ref{lem:agreement} applies
\lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/SwapSubst.lean\#L168}.
\end{proof}

The formalization also proves Crole's general form, for $n$ swaps
applied one after the other, by an induction on $n$ that commutes the
remaining swaps past the first.

\begin{proof}[Proof idea of Theorem~\ref{thm:crole-equivalence}]
Each of (ii)--(v) is shown equivalent to (i) by showing that each of
the two relations is closed under the rules of the other. Only the
rules for abstractions need an argument, and except for the rules
$\alpha$ and $\alpha\#$ the argument follows Crole~\cite[pp.~5--8]{crole2012a}.
A witness that is only fresh (\textsc{pi\#}) or may equal a bound atom
(\textsc{pi1}) is exchanged for an unused one by
Lemmas~\ref{lem:swap-preserve} and~\ref{lem:agreement}. The renaming
relations are closed under \textsc{pi}: Lemma~\ref{lem:swap-subst}
turns the two swaps of its premise into substitutions, and two
instances of $\alpha$ or $\alpha\#$ complete the step.

For the converse, reflexivity, symmetry and transitivity of $\simp$
are already in CSLib, and \textsc{bcg} follows from
Lemma~\ref{lem:swap-preserve} and \textsc{pi}. For $\alpha$ and
$\alpha\#$, let $a' \fresh M$ and choose
$z \nocc a, a', M, \subst{M}{a}{a'}$. Then
$\sw{z}{a} \act M \simp \sw{z}{a'} \act \subst{M}{a}{a'}$ by
Lemma~\ref{lem:swap-chain}, which is the premise of
\textsc{pi} for $\lam{a}{M} \simp \lam{a'}{\subst{M}{a}{a'}}$.
\end{proof}

\subsection{Notes on the formalization}
\label{subsec:formalization-notes}

The formalization follows~\cite{crole2012a} closely, and we found no
errors in it.
Our additions to CSLib are the four relations other than $\simp$,
Theorem~\ref{thm:crole-equivalence},
Proposition~\ref{prop:variants},
Lemmas~\ref{lem:swap-preserve}--\ref{lem:swap-chain} and Crole's
Lemma~6.6. The rest belongs to
the named representation of Section~\ref{sec:cslib}.

Where the formalization departs from~\cite{crole2012a} beyond the
proofs discussed above:
\begin{itemize}
  \item \emph{Order of the results.} Reflexivity, symmetry and
    transitivity of $\simp$ are proved directly. Crole obtains them
    inside the proof of his
    Theorem~4.4~\cite[p.~6]{crole2012a} and cites that theorem for
    them in the proof of Theorem~4.1~\cite[p.~5]{crole2012a}. In the
    formalization Theorem~4.1 does not depend on Theorem~4.4.
  \item \emph{Renaming in \textsc{pi}.} The rules \textsc{pi} and
    \textsc{pi1} are stated in CSLib with renaming
    (Definition~\ref{def:rename}) in place of the swap. Under their
    side conditions the two coincide.
  \item \emph{The rules $\alpha$ and $\alpha\#$.} Crole proves
    $\alpha$ by an induction on $M$~\cite[pp.~6--7]{crole2012a}. For
    $\alpha\#$ this induction fails, because $\subst{M}{a}{a'}$ may
    have to rename a binder, and Crole introduces Lemma~6.5 for
    general $n$ to repair it~\cite[p.~8]{crole2012a}. The formalization derives both rules
    from the case $n = 1$ (Lemma~\ref{lem:swap-chain}).
  \item \emph{Freshness in Lemma~6.5.} Crole assumes $z \nocc M$, but
    freshness suffices. A bound occurrence of $z$ can first be
    renamed away by Crole's Lemma~6.6: if $a \fresh M$ then
    there is a term $\hat{M}$ with $a \nocc \hat{M}$ and
    $\hat{M} \simr M$
    \lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/SwapSubst.lean\#L92}.
    Crole regards this lemma as a form of Barendregt's variable
    convention~\cite[Convention~2.1.13]{barendregt1985a}, \cite[p.~8]{crole2012a}.
  \item \emph{Lemma~6.6 for $\simp$.} Crole states Lemma~6.6 for
    $\simr$ only. Lemma~\ref{lem:swap-chain} needs it for $\simp$
    \lean{\CroleSrc/Named/Untyped/AlphaEquivVariants/SwapSubst.lean\#L116}.
    This form cannot be derived from
    Theorem~\ref{thm:crole-equivalence}, because that theorem's proof
    uses Lemma~\ref{lem:swap-chain}. The proof follows Crole's
    sketch~\cite[p.~16]{crole2012a}, except where the binder is $a$
    itself. There, renaming $a$ to an unused atom throughout the body
    removes every occurrence at once, so no induction on the size of
    the term is needed.
  \item \emph{Permutations.} Lemma~\ref{lem:agreement} is proved for
    arbitrary bijections on $\Atoms$, where Crole assumes finite
    support, a difference that is immaterial for
    terms~\cite[Remark~2.25]{gabbay2011a}. Agreement on a set of atoms is
    stated with Mathlib's \texttt{Set.EqOn}
    \lean{https://github.com/leanprover-community/mathlib4/blob/625d8914199bc567cca9b91f90e8e5afafa5ae51/Mathlib/Data/Set/Operations.lean\#L290}
    instead of a dedicated agreement set definition.
  \item \emph{Scope.} Crole's second contribution, $\alpha$-equivalence of
    program contexts~\cite[Section~5]{crole2012a}, is not covered.
\end{itemize}

that the paper leaves to the reader or to the literature.
